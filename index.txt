function filtered(){
  var s=$("fSes").value,c=$("fCam").value,q=$("fQ").value.trim(),can=$("fCan").checked;
  return adm.rows.filter(function(r){
    return (can||r.status==="접수")&&(!s||r.session===s)&&(!c||r.campus===c)&&(!q||r.name.indexOf(q)>=0);
  });
}
function renderList(){
  var rows=filtered();
  var opts=adm.sessions.map(function(x){return '<option value="'+esc(x.id)+'">'+esc(x.id)+'</option>';}).join("");
  $("tbl").innerHTML='<tr><th>접수번호</th><th>접수시각</th><th>성명</th><th>캠퍼스</th><th>학과</th><th>직급</th><th>이메일</th><th>연락처</th><th>회차</th><th>상태</th><th>관리</th></tr>'+
  (rows.map(function(r){
    var act=r.status==="접수";
    return '<tr'+(act?'':' style="color:#999"')+'><td>'+esc(r.no)+'</td><td>'+esc(r.time)+'</td><td>'+esc(r.name)+'</td><td>'+esc(r.campus)+'</td><td>'+esc(r.dept)+'</td><td>'+esc(r.rank)+'</td><td>'+esc(r.email)+'</td><td>'+esc(r.phone)+'</td><td>'+esc(r.session)+'</td><td>'+esc(r.status)+'</td><td>'+
     (act?'<select data-ch="'+esc(r.no)+'"><option value="">회차변경</option>'+opts+'</select> <button class="b red" data-cn="'+esc(r.no)+'">취소</button>':'')+'</td></tr>';
  }).join("")||'<tr><td colspan="11">해당하는 신청이 없습니다.</td></tr>');
}
$("tbl").addEventListener("click",function(e){
  var no=e.target.getAttribute("data-cn");if(!no)return;
  if(!confirm("접수번호 "+no+" 신청을 취소할까요? 정원에서 제외됩니다."))return;
  post({action:"adminCancel",token:token,no:no}).then(function(r){
    if(r.code==="AUTH")return logout(r.msg);
    alert(r.ok?"취소되었습니다.":r.msg);loadAdmin(false);
  });
});
$("tbl").addEventListener("change",function(e){
  var no=e.target.getAttribute("data-ch");if(!no||!e.target.value)return;
  var ses=e.target.value;
  if(!confirm("접수번호 "+no+"를 "+ses+"로 변경할까요?")){e.target.value="";return;}
  post({action:"adminChange",token:token,no:no,session:ses}).then(function(r){
    if(r.code==="AUTH")return logout(r.msg);
    alert(r.ok?"변경되었습니다.":r.msg);loadAdmin(false);
  });
});

     (rows.map(function(r,i){return '<tr><td>'+(i+1)+'</td><td>'+esc(r.no)+'</td><td>'+esc(r.name)+'</td><td>'+esc(r.campus)+'</td><td>'+esc(r.dept)+'</td><td>'+esc(r.rank)+'</td><td>'+esc(r.phone)+'</td><td></td></tr>';}).join("")||'<tr><td colspan="8">신청자 없음</td></tr>')+

  window.print();
};
$("doCsv").onclick=function(){
  var rows=filtered();
  var head=["접수번호","접수시각","성명","캠퍼스","학과","직급","이메일","연락처","회차","상태"];
  function q(v){return '"'+String(v==null?"":v).replace(/"/g,'""')+'"';}
  var lines=[head.map(q).join(",")].concat(rows.map(function(r){
    return [r.no,r.time,r.name,r.campus,r.dept,r.rank,r.email,r.phone,r.session,r.status].map(q).join(",");}));
  var blob=new Blob(["\uFEFF"+lines.join("\r\n")],{type:"text/csv;charset=utf-8"});
  var a=document.createElement("a");a.href=URL.createObjectURL(blob);a.download="교육신청자명단.csv";
  document.body.appendChild(a);a.click();document.body.removeChild(a);
};
